Load Libraries & Data

#load library
library(tidyverse)
library(ggplot2)
#load data
data <- read.csv("SAFER_RA_cleaned.csv")

Risk Assessment: Number of Different Analytes

#total primary analytes
data %>%
  filter(RISK_ASSESSMENT_RESULT != is.na(RISK_ASSESSMENT_RESULT)) %>%
  ggplot(mapping=aes(x = TOTAL_PRIMARY_ANALYTES, y = RISK_ASSESSMENT_RESULT))+
  geom_boxplot(na.rm = T)+
  labs(title = "Risk Assessment Result vs Total Primary Analytes")

#total secondary analytes
data %>%
  filter(RISK_ASSESSMENT_RESULT != is.na(RISK_ASSESSMENT_RESULT)) %>%
  ggplot(mapping=aes(x = TOTAL_SECONDARY_ANALYTES, y = RISK_ASSESSMENT_RESULT))+
  geom_boxplot(na.rm = T)+
  labs(title = "Risk Assessment Result vs Total Secondary Analytes")

#total e coli analytes
data %>%
  filter(RISK_ASSESSMENT_RESULT != is.na(RISK_ASSESSMENT_RESULT)) %>%
  ggplot(mapping=aes(x = TOTAL_E_COLI_ANALYTES, y = RISK_ASSESSMENT_RESULT))+
  geom_boxplot(na.rm = T)+
  labs(title = "Risk Assessment Result vs Total E. Coli Analytes")

At-Risk Assessment: Types of Different Analytes

data %>%
  filter(PRIMARY_ANALYTE_1 != is.na(PRIMARY_ANALYTE_1) 
         & RISK_ASSESSMENT_RESULT == "At-Risk") %>%
  ggplot(mapping=aes(x = PRIMARY_ANALYTE_1))+
  geom_bar(na.rm = T)+
  coord_flip()+
  labs(title = "At-Risk Assessment: Primary Analyte 1")

data %>%
  filter(PRIMARY_ANALYTE_2 != is.na(PRIMARY_ANALYTE_2) 
         & RISK_ASSESSMENT_RESULT == "At-Risk") %>%
  ggplot(mapping=aes(x = PRIMARY_ANALYTE_2))+
  geom_bar(na.rm = T)+
  coord_flip()+
  labs(title = "At-Risk Assessment: Primary Analyte 2")

data %>%
  filter(PRIMARY_ANALYTE_3 != is.na(PRIMARY_ANALYTE_3) 
         & RISK_ASSESSMENT_RESULT == "At-Risk") %>%
  ggplot(mapping=aes(x = PRIMARY_ANALYTE_3))+
  geom_bar(na.rm = T)+
  coord_flip()+
  labs(title = "At-Risk Assessment: Primary Analyte 3")

It’s pretty obvious the type of analyte matters far more than the number present.
